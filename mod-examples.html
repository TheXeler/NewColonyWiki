脚本

下面是完整的入口脚本，展示了如何订阅事件、设置定时器、监听 Focus 触发。每次加新功能时，可以参考这里的事件名和写法。

local M = {}

function M.on_load()
  engine.log.info("[copper_age] loaded")

  local items = engine.modding.list_by_type("blocks")
  engine.log.info("[copper_age] visible block records: " .. tostring(#items))

  engine.timers.after(20, function(tick)
    engine.log.info("[copper_age] delayed setup tick " .. tostring(tick))
  end)

  engine.focus.subscribe(function(focus)
    if focus.id == "copper_age:copper_found" then
      engine.log.info("[copper_age] focus triggered: " .. focus.title)
    end
  end)

  engine.events.subscribe("inventory:changed", function(evt)
    if evt.item_key == "copper_age:copper_ingot" then
      engine.log.info("[copper_age] ingot inventory changed")
    end
  end)

  engine.events.subscribe("recipe:crafted", function(evt)
    if evt.recipe_id == "copper_age:smelt_copper" then
      engine.log.info("[copper_age] recipe crafted: " .. evt.output_item_key)
    end
  end)

  engine.events.subscribe("profession:assigned", function(evt)
    if evt.profession_id == "copper_age:coppersmith" then
      engine.log.info("[copper_age] coppersmith assigned at level " .. tostring(evt.level))
    end
  end)

  engine.events.subscribe("block:interacted", function(evt)
    engine.log.info("[copper_age] block intent " .. evt.action ..
      " at " .. tostring(evt.x) .. "," .. tostring(evt.y) .. "," .. tostring(evt.z))
  end)

  engine.events.subscribe("focus:chosen", function(evt)
    if evt.focus_id == "copper_age:copper_found" then
      engine.log.info("[copper_age] focus choice selected: " .. evt.choice_id)
    end
  end)

  engine.events.subscribe("focus:dismissed", function(evt)
    if evt.focus_id == "copper_age:copper_found" then
      engine.log.info("[copper_age] focus dismissed")
    end
  end)
end

return M

在控制台中验证

复制 MOD 到 mods/ 下启动游戏，在控制台跑一遍这些命令：

mods
list_data blocks prefix=copper_age:
list_data professions prefix=copper_age:
list_data civilizations prefix=copper_age:
list_data focuses prefix=copper_age:
list_recipes prefix=copper_age:
data copper_age:coppersmith
data copper_age:copper_clan
data copper_age:copper_found
这个示例 MOD 也纳入了自动化加载测试。如果你的 MOD 结构参考它来写，大概率不会遇到加载阶段的问题。
